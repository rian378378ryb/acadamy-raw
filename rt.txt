      parseVideos(videoData) {
        if (!videoData) return [];
        if (Array.isArray(videoData)) return videoData.filter(v => v && v.url && v.url.trim());
        if (typeof videoData === 'string') {
          const trimmed = videoData.trim();
          if (!trimmed) return [];
          if (trimmed.startsWith('[')) {
            try {
              const parsed = JSON.parse(trimmed);
              if (Array.isArray(parsed)) return parsed.filter(v => v && v.url && v.url.trim());
            } catch (e) {}
          }
          return [{ title: 'Tutorial Tool', url: trimmed }];
        }
        return [];
      }

          }
        }
        return trimmed;

      buildVideoEmbed(url) {
        if (!url) {
          return '<div class="text-slate-400 text-sm flex items-center"><i class="fa-solid fa-video-slash mr-2"></i>Video belum disediakan</div>';
        }

        let embedUrl = url;
        if (url.includes('youtube.com/watch?v=')) {
          const vidId = url.split('v=')[1]?.split('&')[0];
          embedUrl = `https://www.youtube.com/embed/${vidId}?autoplay=1&rel=0`;
        } else if (url.includes('youtu.be/')) {
          const vidId = url.split('youtu.be/')[1]?.split('?')[0];
          embedUrl = `https://www.youtube.com/embed/${vidId}?autoplay=1&rel=0`;
        } else if (url.includes('drive.google.com') && url.includes('/view')) {
          embedUrl = url.replace('/view', '/preview');
        }

        return `<iframe src="${embedUrl}" class="w-full h-full border-0" allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture" allowfullscreen></iframe>`;
      }

      handleThumbnailUpload(e) {
        const file = e.target.files[0];
        if (!file) return;

        const statusEl = document.getElementById('materi-thumb-upload-status');
        const inputUrl = document.getElementById('form-materi-thumb-url');
        if (statusEl) statusEl.classList.remove('hidden');

        const reader = new FileReader();
        reader.onload = (uploadEvt) => {
          const img = new Image();
          img.onload = () => {
            const canvas = document.createElement('canvas');
            const MAX_WIDTH = 1200;
            let width = img.width;
            let height = img.height;

            if (width > MAX_WIDTH) {
              height = Math.round((height * MAX_WIDTH) / width);
              width = MAX_WIDTH;
            }

            canvas.width = width;
            canvas.height = height;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(img, 0, 0, width, height);

            const compressedBase64 = canvas.toDataURL('image/jpeg', 0.85);
            this.updateThumbPreview(compressedBase64);

            if (this.isGas) {
              google.script.run
                .withSuccessHandler((res) => {
                  if (statusEl) statusEl.classList.add('hidden');
                  if (res && res.success) {
                    const finalCdn = res.cdnUrl || res.url;
                    inputUrl.value = finalCdn;
                    this.updateThumbPreview(finalCdn);
                    this.showToast('Thumbnail berhasil diupload ke Google Drive!', 'success');
                  } else {
                    this.showToast('Upload gagal: ' + (res?.error || 'Unknown'), 'error');
                  }
                })
                .withFailureHandler((err) => {
                  if (statusEl) statusEl.classList.add('hidden');
                  this.showToast('Gagal upload ke Drive: ' + err.toString(), 'error');
                })
                .uploadFileToDrive(compressedBase64, file.name.replace(/\.[^/.]+$/, "") + ".jpg", "image/jpeg");
            } else if (this.GAS_API_URL && this.GAS_API_URL.startsWith('https://script.google.com')) {
              this.callBackend('uploadFile', {
                base64Data: compressedBase64,
                fileName: file.name.replace(/\.[^/.]+$/, "") + ".jpg",
                mimeType: "image/jpeg"
              }).then((res) => {
                if (statusEl) statusEl.classList.add('hidden');
                if (res && res.success) {
                  const finalCdn = res.cdnUrl || res.url;
                  inputUrl.value = finalCdn;
                  this.updateThumbPreview(finalCdn);
                  this.showToast('Thumbnail berhasil diupload ke Google Drive!', 'success');
                } else {
                  inputUrl.value = compressedBase64;
                  this.showToast('Thumbnail tersimpan (mode demo).', 'success');
                }
              });
            } else {
              if (statusEl) statusEl.classList.add('hidden');
              inputUrl.value = compressedBase64;
              this.showToast('Thumbnail tersimpan (mode offline).', 'success');
            }
          };
          img.src = uploadEvt.target.result;
        };
        reader.readAsDataURL(file);
      }

      async handleSaveMateri(e) {
        e.preventDefault();
        const id = document.getElementById('form-materi-id').value;
        const judul = document.getElementById('form-materi-judul').value.trim();
        const deskripsi = document.getElementById('form-materi-deskripsi').value.trim();
        const thumbnail = document.getElementById('form-materi-thumb-url').value.trim();
        const video_url = document.getElementById('form-materi-video').value.trim();
        const pdf_url = document.getElementById('form-materi-pdf').value.trim();
        const catatan = document.getElementById('form-materi-catatan').value.trim();
        const urutan = parseInt(document.getElementById('form-materi-urutan').value) || 1;
        const status = document.getElementById('form-materi-status').value;

        const payload = {
          id: id || ('MAT' + String(Date.now()).slice(-5)),
          judul, deskripsi, thumbnail, video_url, pdf_url, catatan, urutan, status,
          created_at: new Date().toISOString().replace('T', ' ').slice(0, 19)
        };

        if (id) {
          const index = this.materi.findIndex(m => m.id === id);
          if (index !== -1) this.materi[index] = payload;
        } else {
          this.materi.push(payload);
        }

        this.syncStorage();

        if (this.isGas) {
          google.script.run
            .withSuccessHandler(() => {
              this.showToast('Materi tersimpan ke Google Sheets.', 'success');
            })
            .saveMateri(payload);
        } else if (this.GAS_API_URL && this.GAS_API_URL.startsWith('https://script.google.com')) {
          await this.callBackend('saveMateri', { data: payload });
          this.showToast('Materi tersimpan ke Google Sheets.', 'success');
        } else {
          this.showToast('Materi berhasil disimpan.', 'success');
        }

        this.closeModal('modal-materi-form');
        this.renderAdminMateri();
        this.renderAdminOverview();
      }

        }
      }

        if (fileNameEl) fileNameEl.textContent = file.name;
        if (statusEl) statusEl.classList.remove('hidden');

        const reader = new FileReader();
        reader.onload = (uploadEvt) => {
          const base64Data = uploadEvt.target.result;

          if (this.isGas) {
            google.script.run
              .withSuccessHandler((res) => {
                if (statusEl) statusEl.classList.add('hidden');
                if (res && res.success) {
                  const finalUrl = res.previewUrl || res.url;
                  if (inputPdfUrl) inputPdfUrl.value = finalUrl;
                  if (inputDirectUrl) inputDirectUrl.value = finalUrl;
                  if (jenisSelect) jenisSelect.value = 'PDF';
                  if (iconInput && (!iconInput.value || iconInput.value === 'fa-solid fa-wrench')) {
                    iconInput.value = 'fa-solid fa-file-pdf';
                  }
                  this.showToast('File PDF berhasil diupload ke Google Drive!', 'success');
                } else {
                  this.showToast('Upload gagal: ' + (res?.error || 'Unknown'), 'error');
                }
              })
              .withFailureHandler((err) => {
                if (statusEl) statusEl.classList.add('hidden');
                this.showToast('Gagal upload ke Drive: ' + err.toString(), 'error');
              })
              .uploadFileToDrive(base64Data, file.name, file.type);
          } else {
            setTimeout(() => {
              if (statusEl) statusEl.classList.add('hidden');
              const dummyUrl = 'https://www.w3.org/WAI/ER/tests/xhtml/testfiles/resources/pdf/dummy.pdf';
              if (inputPdfUrl) inputPdfUrl.value = dummyUrl;
              if (inputDirectUrl) inputDirectUrl.value = dummyUrl;
              if (jenisSelect) jenisSelect.value = 'PDF';
              if (iconInput && (!iconInput.value || iconInput.value === 'fa-solid fa-wrench')) {
                iconInput.value = 'fa-solid fa-file-pdf';
              }
              this.showToast('File PDF terpasang (mode demo).', 'success');
            }, 500);
          }
        };
        reader.readAsDataURL(file);
      }

      async handleSaveResource(e) {
        e.preventDefault();
        const id = document.getElementById('form-res-id').value;
        const materi_id = document.getElementById('form-res-materi-id').value;
        const nama = document.getElementById('form-res-nama').value.trim();
        const jenis = document.getElementById('form-res-jenis').value;
        
        let url = '';
        if (this.currentResourceInputMode === 'PDF') {
          url = document.getElementById('form-res-pdf-url').value.trim() || document.getElementById('form-res-url').value.trim();
        } else {
          url = document.getElementById('form-res-url').value.trim() || document.getElementById('form-res-pdf-url').value.trim();
        }

        if (!url) {
          this.showToast('Mohon masukkan link URL atau upload file PDF terlebih dahulu.', 'warning');
          return;
        }

        const videoRows = document.querySelectorAll('#res-videos-container .res-video-row');
        const collectedVideos = [];
        videoRows.forEach((row, idx) => {
          const tInput = row.querySelector('.res-video-title');
          const uInput = row.querySelector('.res-video-url');
          const vUrl = uInput ? uInput.value.trim() : '';
          const vTitle = (tInput && tInput.value.trim()) ? tInput.value.trim() : `Part ${idx + 1}`;
          if (vUrl) {
            collectedVideos.push({ title: vTitle, url: vUrl });
          }
        });

        const video_url = collectedVideos.length > 0 ? JSON.stringify(collectedVideos) : '';

        const deskripsi = document.getElementById('form-res-deskripsi').value.trim();
        const icon = document.getElementById('form-res-icon').value.trim() || (jenis === 'PDF' ? 'fa-solid fa-file-pdf' : 'fa-solid fa-wrench');
        const urutan = parseInt(document.getElementById('form-res-urutan').value) || 1;
        const status = document.getElementById('form-res-status').value;

        const payload = {
          id: id || ('RES' + String(Date.now()).slice(-5)),
          materi_id, nama, jenis, url, video_url, deskripsi, icon, urutan, status
        };

        if (id) {
          const index = this.resources.findIndex(r => r.id === id);
          if (index !== -1) this.resources[index] = { ...this.resources[index], ...payload };
        } else {
          this.resources.push(payload);
        }

        this.syncStorage();

        if (this.isGas) {
          google.script.run
            .withSuccessHandler(() => {
              this.showToast('Resource tersimpan ke Google Sheets.', 'success');
            })
            .saveResource(payload);
        } else if (this.GAS_API_URL && this.GAS_API_URL.startsWith('https://script.google.com')) {
          await this.callBackend('saveResource', { data: payload });
          this.showToast('Resource tersimpan ke Google Sheets.', 'success');
        } else {
          this.showToast('Resource berhasil disimpan.', 'success');
        }

        this.closeModal('modal-resource-form');
        this.renderAdminResources();
        this.renderAdminOverview();
      }

      async toggleResourceStatus(id) {
        const r = this.resources.find(x => x.id === id);
        if (r) {
          r.status = r.status === 'ACTIVE' ? 'INACTIVE' : 'ACTIVE';
          this.syncStorage();
          if (this.isGas) google.script.run.saveResource(r);
          else if (this.GAS_API_URL && this.GAS_API_URL.startsWith('https://script.google.com')) {
            await this.callBackend('saveResource', { data: r });
          }
          this.renderAdminResources();
          this.showToast(`Status resource diubah menjadi ${r.status}`, 'info');
        }
      }

      confirmDeleteResource(id) {
        this.showConfirm(
          'Hapus Resource Ini?',
          'Resource ini tidak akan muncul lagi di halaman materi member.',
          async () => {
            this.resources = this.resources.filter(r => r.id !== id);
            this.syncStorage();
            if (this.isGas) google.script.run.deleteResource(id);
            else if (this.GAS_API_URL && this.GAS_API_URL.startsWith('https://script.google.com')) {
              await this.callBackend('deleteResource', { id });
            }
            this.showToast('Resource berhasil dihapus.', 'info');
            this.renderAdminResources();
            this.renderAdminOverview();
          }
        );
      }

      showToast(message, type = 'info') {
        const container = document.getElementById('toast-container');
        if (!container) return;

        const toast = document.createElement('div');
        toast.className = 'pointer-events-auto px-4 py-3 rounded-2xl shadow-xl flex items-center space-x-3 text-xs sm:text-sm font-semibold transition-all duration-300 transform translate-y-3 opacity-0';

        let icon = '<i class="fa-solid fa-circle-info text-blue-600"></i>';
        let bg = 'bg-white border border-slate-200 text-slate-800';

        if (type === 'success') {
          icon = '<i class="fa-solid fa-circle-check text-emerald-600"></i>';
          bg = 'bg-white border border-emerald-300 text-slate-800';
        } else if (type === 'error') {
          icon = '<i class="fa-solid fa-circle-xmark text-rose-600"></i>';
          bg = 'bg-white border border-rose-300 text-slate-800';
        } else if (type === 'warning') {
          icon = '<i class="fa-solid fa-triangle-exclamation text-amber-600"></i>';
          bg = 'bg-white border border-amber-300 text-slate-800';
        }

        toast.innerHTML = `${icon}<span>${this.escapeHtml(message)}</span>`;
        toast.className += ` ${bg}`;
        container.appendChild(toast);

        requestAnimationFrame(() => {
          toast.classList.remove('translate-y-3', 'opacity-0');
        });

        setTimeout(() => {
          toast.classList.add('opacity-0', 'translate-y-3');
          setTimeout(() => toast.remove(), 300);
        }, 4000);
      }

      showConfirm(title, desc, onConfirm) {
        const modal = document.getElementById('modal-confirm');
        const titleEl = document.getElementById('confirm-title');
        const descEl = document.getElementById('confirm-desc');
        const btnCancel = document.getElementById('btn-confirm-cancel');
        const btnProceed = document.getElementById('btn-confirm-proceed');

        if (!modal) return;

        titleEl.textContent = title;
        descEl.textContent = desc;

        modal.classList.remove('hidden');

        btnCancel.onclick = () => modal.classList.add('hidden');
        btnProceed.onclick = () => {
          modal.classList.add('hidden');
          if (onConfirm) onConfirm();
        };
      }

      escapeHtml(str) {
        if (!str) return '';
        return String(str)
          .replace(/&/g, '&amp;')
          .replace(/</g, '&lt;')
          .replace(/>/g, '&gt;')
          .replace(/"/g, '&quot;')
          .replace(/'/g, '&#039;');
      }
    }
